-- 1. LIMPIEZA INICIAL
DROP TABLE IF EXISTS AuditoriaAcceso, ReporteMedico, Equipamiento, Cita, Prescripcion, HistoriaClinica, Provee, Proveedor, Medicamento, Empleado, Departamento, Rol, Cargo, Paciente, Persona, TipoDocumento, Genero, Sede, TipoReporte, Estado, Tipo_servicio CASCADE;

-- 2. EXTENSIONES Y SEGURIDAD
CREATE EXTENSION IF NOT EXISTS dblink;
CREATE EXTENSION IF NOT EXISTS pgcrypto; -- Para encriptación de contraseñas

-- 3. MODELO DE DATOS (CREATE TABLE) - MISMO ESQUEMA

CREATE TABLE Sede (
  id_sede INTEGER PRIMARY KEY,
  nombre_sede VARCHAR(50) NOT NULL
);

CREATE TABLE Genero (
  cod_genero INTEGER PRIMARY KEY,
  nom_genero VARCHAR(20) NOT NULL
);

CREATE TABLE TipoDocumento (
  id_tipo_doc INTEGER PRIMARY KEY,
  nom_tipo_doc VARCHAR(20) NOT NULL
);

CREATE TABLE Persona (
  documento INTEGER PRIMARY KEY,
  nombre_completo VARCHAR(50) NOT NULL,
  telefono VARCHAR(20) NOT NULL,
  correo VARCHAR(50) NOT NULL,
  id_genero INTEGER NOT NULL,
  id_tipo_doc INTEGER NOT NULL,
  id_sede_origen INTEGER NOT NULL, -- para fragmentación por sede
  FOREIGN KEY (id_genero) REFERENCES Genero(cod_genero),
  FOREIGN KEY (id_tipo_doc) REFERENCES TipoDocumento(id_tipo_doc),
  FOREIGN KEY (id_sede_origen) REFERENCES Sede(id_sede) -- Sede debe existir en maestros
);

CREATE TABLE Paciente (
  cod_pac INTEGER PRIMARY KEY,
  direccion VARCHAR(50) NOT NULL,
  documento INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,             
  FOREIGN KEY (documento) REFERENCES Persona(documento),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE Cargo (
  id_cargo INTEGER PRIMARY KEY,
  nombre_cargo VARCHAR(50) NOT NULL
);

CREATE TABLE Rol (
  id_rol INTEGER PRIMARY KEY,
  nombre_rol VARCHAR(50) NOT NULL
);



CREATE TABLE Departamento (
  id_dept INTEGER NOT NULL,
  nom_dept VARCHAR(50) NOT NULL,
  id_sede INTEGER NOT NULL,
  PRIMARY KEY (id_dept, id_sede),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE Empleado (
  id_emp INTEGER PRIMARY KEY,
  hash_contra VARCHAR(200) NOT NULL, 
  documento INTEGER NOT NULL,
  id_cargo INTEGER NOT NULL,
  id_rol INTEGER NOT NULL,
  id_dept INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,
  FOREIGN KEY (documento) REFERENCES Persona(documento),
  FOREIGN KEY (id_cargo) REFERENCES Cargo(id_cargo),
  FOREIGN KEY (id_rol) REFERENCES Rol(id_rol),
  FOREIGN KEY (id_dept, id_sede) REFERENCES Departamento(id_dept, id_sede),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE Medicamento (
  cod_med INTEGER PRIMARY KEY,
  nom_med VARCHAR(50) NOT NULL,
  descripcion VARCHAR(50) NOT NULL,
  stock INTEGER NOT NULL CHECK (stock > 0),
  unidad VARCHAR(50) NOT NULL
);

CREATE TABLE Proveedor (
  cod_prov INTEGER PRIMARY KEY,
  nombre_prov VARCHAR(50) NOT NULL
);

CREATE TABLE Provee (
  cod_prov INTEGER NOT NULL,
  cod_med INTEGER NOT NULL,
  PRIMARY KEY (cod_prov, cod_med),
  FOREIGN KEY (cod_prov) REFERENCES Proveedor(cod_prov),
  FOREIGN KEY (cod_med) REFERENCES Medicamento(cod_med)
);

CREATE TABLE HistoriaClinica (
  cod_hist INTEGER PRIMARY KEY,
  fecha_registro DATE NOT NULL,
  diagnostico VARCHAR(200),
  cod_medico INTEGER NOT NULL,
  cod_pac INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,
  FOREIGN KEY (cod_medico) REFERENCES Empleado(id_emp),
  FOREIGN KEY (cod_pac) REFERENCES Paciente(cod_pac),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE Prescripcion (
  id_presc INTEGER PRIMARY KEY,
  dosis VARCHAR(50) NOT NULL,
  frecuencia VARCHAR(50) NOT NULL,
  duracion VARCHAR(50) NOT NULL,
  fecha_emision DATE NOT NULL,
  cod_med INTEGER,  
  cod_prov INTEGER, 
  cod_hist INTEGER NOT NULL,
  FOREIGN KEY (cod_prov, cod_med) REFERENCES Provee(cod_prov, cod_med),
  FOREIGN KEY (cod_hist) REFERENCES HistoriaClinica(cod_hist)
);

CREATE TABLE Tipo_servicio (
  id_servicio INTEGER PRIMARY KEY,
  nom_servicio VARCHAR(50) NOT NULL
);

CREATE TABLE Estado (
  id_estado INTEGER PRIMARY KEY,
  nom_estado VARCHAR(50) NOT NULL
);

CREATE TABLE Cita (
  id_cita INTEGER PRIMARY KEY,
  fecha DATE NOT NULL,
  hora TIME NOT NULL,
  id_servicio INTEGER NOT NULL,
  id_estado INTEGER NOT NULL,
  id_emp INTEGER NOT NULL,
  id_dept INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,
  FOREIGN KEY (id_servicio) REFERENCES Tipo_servicio(id_servicio),
  FOREIGN KEY (id_estado) REFERENCES Estado(id_estado),
  FOREIGN KEY (id_emp) REFERENCES Empleado(id_emp),
  FOREIGN KEY (id_dept, id_sede) REFERENCES Departamento(id_dept, id_sede),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE Equipamiento (
  cod_eq INTEGER PRIMARY KEY,
  nom_eq VARCHAR(50) NOT NULL,
  fecha_mantenimiento DATE,
  estado VARCHAR(50) NOT NULL,
  responsable INTEGER NOT NULL,
  id_dept INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,
  FOREIGN KEY (responsable) REFERENCES Empleado(id_emp),
  FOREIGN KEY (id_dept, id_sede) REFERENCES Departamento(id_dept, id_sede),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE TipoReporte (
  id_tipo_reporte INTEGER PRIMARY KEY,
  nombre_tipo_reporte VARCHAR(50) NOT NULL
);

CREATE TABLE ReporteMedico (
  id_reporte INTEGER PRIMARY KEY,
  fecha_generacion DATE NOT NULL,
  resumen VARCHAR(200) NOT NULL,
  id_tipo_reporte INTEGER NOT NULL,
  id_sede INTEGER NOT NULL,
  FOREIGN KEY (id_tipo_reporte) REFERENCES TipoReporte(id_tipo_reporte),
  FOREIGN KEY (id_sede) REFERENCES Sede(id_sede)
);

CREATE TABLE AuditoriaAcceso (
  id_evento INTEGER PRIMARY KEY,
  accion VARCHAR(50) NOT NULL,
  tabla_afectada VARCHAR(50) NOT NULL,
  fecha_evento TIMESTAMP NOT NULL DEFAULT now(),
  ip_origen VARCHAR(50) NOT NULL,
  id_emp INTEGER NOT NULL,
  FOREIGN KEY (id_emp) REFERENCES Empleado(id_emp)
);

-- 9. VISTA DISTRIBUIDA Y AGREGACIÓN
-- VISTA DE MEDICAMENTOS GLOBAL (Simulación de Replicación COMPLETA)
CREATE OR REPLACE VIEW Medicamentos_Global AS
SELECT * FROM Medicamento
UNION ALL
SELECT * FROM dblink('clinica_norte', 'SELECT cod_med, nom_med, descripcion, stock, unidad FROM Medicamento') AS t1(cod_med int, nom_med varchar, descripcion varchar, stock int, unidad varchar)
UNION ALL
SELECT * FROM dblink('hospital_sur', 'SELECT cod_med, nom_med, descripcion, stock, unidad FROM Medicamento') AS t2(cod_med int, nom_med varchar, descripcion varchar, stock int, unidad varchar);


-- VISTA DE HISTORIAS CLÍNICAS GLOBAL (Simulación de Replicación PARCIAL)
CREATE OR REPLACE VIEW HistoriaClinica_Global AS
SELECT * FROM HistoriaClinica -- Datos locales (Sede 1)
UNION ALL
SELECT * FROM dblink('clinica_norte', 'SELECT cod_hist, fecha_registro, diagnostico, cod_medico, cod_pac, id_sede FROM HistoriaClinica') AS t1(cod_hist int, fecha_registro date, diagnostico varchar, cod_medico int, cod_pac int, id_sede int)
UNION ALL
SELECT * FROM dblink('hospital_sur', 'SELECT cod_hist, fecha_registro, diagnostico, cod_medico, cod_pac, id_sede FROM HistoriaClinica') AS t2(cod_hist int, fecha_registro date, diagnostico varchar, cod_medico int, cod_pac int, id_sede int);


-- CONSULTA AGREGADA (Requiere vistas globales para datos entre sedes)
-- Top 5 enfermedades más tratadas por departamento en el último trimestre (Simulado)
CREATE OR REPLACE VIEW Top_Enfermedades_QTR AS
SELECT 
    d.nom_dept,
    h.diagnostico AS enfermedad,
    COUNT(h.cod_hist) AS total_casos,
    s.nombre_sede
FROM 
    HistoriaClinica_Global h
JOIN 
    Empleado e ON h.cod_medico = e.id_emp
JOIN 
    Departamento d ON e.id_dept = d.id_dept AND e.id_sede = d.id_sede
JOIN
    Sede s ON h.id_sede = s.id_sede
WHERE 
    h.fecha_registro >= date_trunc('quarter', CURRENT_DATE - interval '3 month')
GROUP BY 1, 2, 4
ORDER BY total_casos DESC
LIMIT 5;
